  <script>
    /* --- テーマ --- */
    function toggleTheme() {
      const isDark = document.documentElement.classList.toggle('dark');
      localStorage.setItem('theme', isDark ? 'dark' : 'light');
    }
    if (localStorage.getItem('theme') === 'light') {
      document.documentElement.classList.remove('dark');
    }

    /* ===================================================================
       電卓ロジック（元コードから一切変更なし・完全維持）
       'A' 押下時に expression が SECRET_CODE と一致すれば unlockApp()（script1側）
       =================================================================== */
    const SECRET_CODE = '7895123';
    let expression = '';

    function calcInput(value) {
      const display = document.getElementById('calc-display');
      if (value === 'C') { expression = ''; display.value = '0'; return; }
      if (value === 'A') { if (expression === SECRET_CODE) { unlockApp(); expression = ''; display.value = '0'; } return; }
      if (value === '=') {
        if (!expression) return;
        try {
          let parsed = expression
            .replace(/π/g, 'Math.PI')
            .replace(/e/g, 'Math.E')
            .replace(/sin\(/g, 'Math.sin(')
            .replace(/cos\(/g, 'Math.cos(')
            .replace(/tan\(/g, 'Math.tan(')
            .replace(/log\(/g, 'Math.log10(')
            .replace(/ln\(/g, 'Math.log(')
            .replace(/sqrt\(/g, 'Math.sqrt(')
            .replace(/\^/g, '**');

          const result = Function(`'use strict'; return (${parsed})`)();
          if (!Number.isFinite(result)) throw new Error();
          display.value = String(result);
          expression = String(result);
        } catch {
          display.value = 'Error';
          expression = '';
        }
        return;
      }
      if (display.value === '0' || display.value === 'Error') expression = value;
      else expression += value;
      display.value = expression;
    }

    document.addEventListener('keydown', (e) => {
      // プレイヤーモード表示中（ダッシュボード非表示）は電卓のキー操作を無効化
      const dash = document.getElementById('tools-dashboard');
      if (dash && dash.style.display === 'none') return;
      // 入力欄フォーカス中は無効（タイマーの分秒入力など）
      const tag = (document.activeElement && document.activeElement.tagName) || '';
      if (tag === 'INPUT' || tag === 'SELECT' || tag === 'TEXTAREA') return;

      if ((e.key >= '0' && e.key <= '9') || ['.', '+', '-', '*', '/', '(', ')', '^'].includes(e.key)) {
        calcInput(e.key);
      } else if (e.key === 'Enter') {
        calcInput('=');
      } else if (e.key === 'Backspace') {
        expression = expression.slice(0, -1);
        document.getElementById('calc-display').value = expression || '0';
      } else if (e.key === 'Escape') {
        calcInput('C');
      } else if (e.key.toLowerCase() === 'p') {
        calcInput('π');
      } else if (e.key.toLowerCase() === 'e') {
        calcInput('e');
      } else if (e.key.toLowerCase() === 'a') {
        calcInput('A');
      }
    });

    /* --- 通知音 --- */
    function beep() {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const o = ctx.createOscillator(), g = ctx.createGain();
        o.connect(g); g.connect(ctx.destination);
        o.type = 'sine'; o.frequency.value = 880;
        g.gain.setValueAtTime(0.2, ctx.currentTime);
        g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.6);
        o.start(); o.stop(ctx.currentTime + 0.6);
      } catch (e) {}
    }

    /* --- 時計 --- */
    function initClock() {
      const t = document.getElementById('clock-time');
      const d = document.getElementById('clock-date');
      function upd() {
        const now = new Date();
        t.textContent = now.toLocaleTimeString('ja-JP', { hour12: false });
        d.textContent = now.toLocaleDateString('ja-JP', { year: 'numeric', month: 'long', day: 'numeric', weekday: 'short' });
      }
      upd(); setInterval(upd, 1000);
    }

    /* --- カレンダー --- */
    let calDate = new Date();
    function renderCalendar() {
      const y = calDate.getFullYear(), m = calDate.getMonth();
      document.getElementById('cal-title').textContent = `${y}年${m + 1}月`;
      const first = new Date(y, m, 1).getDay();
      const days = new Date(y, m + 1, 0).getDate();
      const today = new Date();
      const grid = document.getElementById('cal-grid');
      grid.innerHTML = '';
      ['日', '月', '火', '水', '木', '金', '土'].forEach((w, i) => {
        const c = document.createElement('div');
        c.textContent = w;
        c.className = 'text-center text-[10px] font-bold pb-0.5 ' + (i === 0 ? 'text-red-500' : i === 6 ? 'text-blue-500' : 'text-gray-400');
        grid.appendChild(c);
      });
      for (let i = 0; i < first; i++) grid.appendChild(document.createElement('div'));
      for (let d = 1; d <= days; d++) {
        const c = document.createElement('div');
        c.textContent = d;
        const isToday = d === today.getDate() && m === today.getMonth() && y === today.getFullYear();
        c.className = 'text-center text-xs py-1 rounded-md ' + (isToday ? 'bg-blue-600 text-white font-bold' : 'hover:bg-gray-100 dark:hover:bg-zinc-700 cursor-default');
        grid.appendChild(c);
      }
    }
    function calPrev() { calDate.setMonth(calDate.getMonth() - 1); renderCalendar(); }
    function calNext() { calDate.setMonth(calDate.getMonth() + 1); renderCalendar(); }

    /* --- カウンター --- */
    let counterVal = 0;
    function counterChange(d) { counterVal += d; document.getElementById('counter-val').textContent = counterVal; }
    function counterReset() { counterVal = 0; document.getElementById('counter-val').textContent = 0; }

    /* --- タイマー（カウントダウン） --- */
    let timerRemaining = 0, timerId = null, timerRunning = false;
    function timerRender() {
      const m = String(Math.floor(timerRemaining / 60)).padStart(2, '0');
      const s = String(timerRemaining % 60).padStart(2, '0');
      document.getElementById('timer-display').textContent = `${m}:${s}`;
    }
    function timerStart() {
      if (timerRunning) return;
      if (timerRemaining <= 0) {
        const mm = parseInt(document.getElementById('timer-min').value) || 0;
        const ss = parseInt(document.getElementById('timer-sec').value) || 0;
        timerRemaining = mm * 60 + ss;
      }
      if (timerRemaining <= 0) return;
      timerRunning = true;
      timerRender();
      timerId = setInterval(() => {
        timerRemaining--;
        timerRender();
        if (timerRemaining <= 0) { timerStop(); beep(); }
      }, 1000);
    }
    function timerStop() { clearInterval(timerId); timerRunning = false; }
    function timerReset() { timerStop(); timerRemaining = 0; timerRender(); }

    /* --- ストップウォッチ --- */
    let swAccum = 0, swBase = 0, swId = null, swRunning = false;
    function swCurrent() { return swAccum + (swRunning ? Date.now() - swBase : 0); }
    function swRender() {
      const t = swCurrent();
      const m = String(Math.floor(t / 60000)).padStart(2, '0');
      const s = String(Math.floor((t % 60000) / 1000)).padStart(2, '0');
      const cs = String(Math.floor((t % 1000) / 10)).padStart(2, '0');
      document.getElementById('sw-display').textContent = `${m}:${s}.${cs}`;
    }
    function swToggle() {
      if (swRunning) { swAccum = swCurrent(); swRunning = false; clearInterval(swId); }
      else { swBase = Date.now(); swRunning = true; swId = setInterval(swRender, 33); }
      document.getElementById('sw-toggle').textContent = swRunning ? '停止' : '開始';
    }
    function swReset() {
      swRunning = false; clearInterval(swId); swAccum = 0; swRender();
      document.getElementById('sw-toggle').textContent = '開始';
    }

    /* --- ポモドーロ --- */
    const POMO_WORK = 25 * 60, POMO_BREAK = 5 * 60;
    let pomoMode = 'work', pomoRemaining = POMO_WORK, pomoId = null, pomoRunning = false, pomoCount = 0;
    function pomoRender() {
      const m = String(Math.floor(pomoRemaining / 60)).padStart(2, '0');
      const s = String(pomoRemaining % 60).padStart(2, '0');
      document.getElementById('pomo-display').textContent = `${m}:${s}`;
      document.getElementById('pomo-mode').textContent = pomoMode === 'work' ? '作業' : '休憩';
      document.getElementById('pomo-count').textContent = `完了: ${pomoCount}`;
    }
    function pomoToggle() {
      if (pomoRunning) { pomoRunning = false; clearInterval(pomoId); }
      else {
        pomoRunning = true;
        pomoId = setInterval(() => {
          pomoRemaining--;
          if (pomoRemaining < 0) {
            beep();
            if (pomoMode === 'work') { pomoCount++; pomoMode = 'break'; pomoRemaining = POMO_BREAK; }
            else { pomoMode = 'work'; pomoRemaining = POMO_WORK; }
          }
          pomoRender();
        }, 1000);
      }
      document.getElementById('pomo-toggle').textContent = pomoRunning ? '一時停止' : '開始';
    }
    function pomoReset() {
      pomoRunning = false; clearInterval(pomoId);
      pomoMode = 'work'; pomoRemaining = POMO_WORK; pomoCount = 0; pomoRender();
      document.getElementById('pomo-toggle').textContent = '開始';
    }

    /* --- 初期化 --- */
    initClock();
    renderCalendar();
    timerRender();
    swRender();
    pomoRender();
    lucide.createIcons();
  </script>
